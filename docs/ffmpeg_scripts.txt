// Works -t before the input file customizes the amount of duration taken from the input file
ffmpeg \
	-t 00:00:05 \
 	-i sw_imperial_march.mp3 \
    -i sw_chewy_roar.mp3 \
    -filter_complex '[1:a] adelay=2500|2500 [delayed]; [0:a] [delayed] amix=inputs=2:duration=longest:dropout_transition=3' \
    out.mp3
